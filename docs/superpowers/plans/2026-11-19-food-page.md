# Food Page Build-Out Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Food page's three placeholders ("Featured", "map", "sort btns go here") with a random featured restaurant, an interactive pin map, and sort and filter controls, plus loading, error, and empty states.

**Architecture:** The Vuex `restaurants` module becomes the single source of truth for the list, filters, open/closed status, and the featured pick. Components read it and dispatch to it. A small `openingHours.js` module wraps the `opening_hours` library and evaluates hours in the searched city's timezone. The map is a `<script setup>` component that holds its Leaflet instance outside Vue reactivity.

**Tech Stack:** Vue 3.5 (Options API, plus `<script setup>` for the map), Vuex 4, Vite 8, Vitest 5 + jsdom, Leaflet 1.9.4, opening_hours 3.16.0, SCSS with shared variables injected by `vite.config.js`.

**Spec:** `docs/superpowers/specs/2026-11-19-food-page-design.md`

Create `src/openingHours.js`:

```js
import OpeningHours from 'opening_hours';

// opening_hours evaluates a Date by its local fields (getHours, getDay...).
// Build a Date whose local fields equal the wall-clock time in `timeZone`,
// so a Denver restaurant is checked against Denver time wherever the viewer is.
function wallClock(now, timeZone) {
  if (!timeZone) return now;
  const parts = Object.fromEntries(
    new Intl.DateTimeFormat('en-US', {
      timeZone,
      hourCycle: 'h23',
      year: 'numeric',
      month: 'numeric',
      day: 'numeric',
      hour: 'numeric',
      minute: 'numeric',
      second: 'numeric',
    })
      .formatToParts(now)
      .map(({ type, value }) => [type, value])
  );
  return new Date(+parts.year, +parts.month - 1, +parts.day, +parts.hour, +parts.minute, +parts.second);
}

// Parsing is the slow part; the same strings repeat across renders.
// ponytail: unbounded cache, fine for a 40-item list per search; add an LRU if it grows.
const cache = new Map();

function parse(hours, lat, lon) {
  const key = `${hours}|${lat}|${lon}`;
  if (!cache.has(key)) {
    let parsed = null;
    try {
      // lat/lon/country let the library resolve sunrise/sunset and US public holidays (PH)
      parsed = new OpeningHours(hours, { lat, lon, address: { country_code: 'us' } });
    } catch {
      parsed = null; // unparseable OSM hours are "unknown", not an error
    }
    cache.set(key, parsed);
  }
  return cache.get(key);
}

/**
 * @param {string} hours OSM opening_hours value, e.g. "Mo-Fr 09:00-17:00"
 * @param {{ timezone?: string|null, lat?: number, lon?: number }} place
 * @param {Date} now
 * @returns {true|false|null} null when the hours are missing, unparseable, or unknown right now
 */
export function isOpenNow(hours, { timezone = null, lat = 0, lon = 0 } = {}, now = new Date()) {
  if (!hours) return null;
  const parsed = parse(hours, lat, lon);
  if (!parsed) return null;
  const local = wallClock(now, timezone);
  if (parsed.getUnknown(local)) return null;
  return parsed.getState(local);
}
```

import apiClient from '@/apiClient';
import { isOpenNow } from '@/openingHours';

const getDefaultFilters = () => ({
  sort: 'distance', // 'distance' | 'name'
  openNow: false,
  cuisine: null,
});

const getDefaultState = () => ({
  restaurants: [], // nearest first, as returned by apiClient
  isLoading: false,
  error: null,
  featuredId: null,
  ...getDefaultFilters(),
});

const pickRandom = (list) => list[Math.floor(Math.random() * list.length)];

export default {
  namespaced: true,
  // function, not object: each store instance (and each test) gets fresh state
  state: getDefaultState,
  mutations: {
    SET_RESTAURANTS(state, value) {
      state.restaurants = value;
    },
    SET_IS_LOADING(state, value) {
      state.isLoading = value;
    },
    SET_ERROR(state, value) {
      state.error = value;
    },
    SET_SORT(state, value) {
      state.sort = value;
    },
    SET_OPEN_NOW(state, value) {
      state.openNow = value;
    },
    SET_CUISINE(state, value) {
      state.cuisine = value;
    },
    CLEAR_FILTERS(state) {
      Object.assign(state, getDefaultFilters());
    },
    SET_FEATURED_ID(state, value) {
      state.featuredId = value;
    },
    RESET_DATA(state) {
      Object.assign(state, getDefaultState());
    },
  },
  getters: {
    restaurants: (state) => state.restaurants,
    isLoading: (state) => state.isLoading,
    error: (state) => state.error,
    sort: (state) => state.sort,
    openNow: (state) => state.openNow,
    cuisine: (state) => state.cuisine,
    // ponytail: cached until the list or location changes, so status can go stale
    // on a page left open for hours; add a minute ticker in state if that matters.
    openStatus(state, _getters, _rootState, rootGetters) {
      const { timezone } = rootGetters['location/location'];
      const now = new Date();
      const label = { true: 'open', false: 'closed', null: 'unknown' };
      return Object.fromEntries(
        state.restaurants.map((r) => [r.id, label[isOpenNow(r.hours, { timezone, lat: r.lat, lon: r.lon }, now)]])
      );
    },
    filteredRestaurants(state, getters) {
      let list = state.restaurants;
      if (state.cuisine) list = list.filter((r) => r.cuisine === state.cuisine);
      if (state.openNow) list = list.filter((r) => getters.openStatus[r.id] === 'open');
      if (state.sort === 'name') list = [...list].sort((a, b) => a.name.localeCompare(b.name));
      return list;
    },
    cuisines(state) {
      const counts = {};
      state.restaurants.forEach(({ cuisine }) => {
        if (cuisine !== 'restaurant') counts[cuisine] = (counts[cuisine] || 0) + 1;
      });
      return Object.entries(counts)
        .sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]))
        .slice(0, 8)
        .map(([name, count]) => ({ name, count }));
    },
    // looked up in the full list so a filter change never leaves a dangling id
    featured: (state) => state.restaurants.find((r) => r.id === state.featuredId) || null,
  },
  actions: {
    async getRestaurants({ commit, dispatch }, coordinates) {
      commit('SET_IS_LOADING', true);
      commit('SET_ERROR', null);
      commit('CLEAR_FILTERS');
      commit('SET_FEATURED_ID', null);
      try {
        commit('SET_RESTAURANTS', await apiClient.getRestaurants(coordinates));
        dispatch('pickInitialFeatured');
      } catch (error) {
        commit('SET_RESTAURANTS', []);
        commit('SET_ERROR', error);
      } finally {
        commit('SET_IS_LOADING', false);
      }
    },
    pickInitialFeatured({ state, getters, commit }) {
      const open = state.restaurants.filter((r) => getters.openStatus[r.id] === 'open');
      const pool = open.length ? open : state.restaurants;
      commit('SET_FEATURED_ID', pool.length ? pickRandom(pool).id : null);
    },

    },

      commit('SET_SORT', value);
    },
    setOpenNow({ commit }, value) {
      commit('SET_OPEN_NOW', value);
    },
    setCuisine({ commit }, value) {
      commit('SET_CUISINE', value);
    },

      commit('CLEAR_FILTERS');
    },
    resetState: {
      root: true,
      handler({ commit }) {

      },
    },

<script setup>
import { computed, onBeforeUnmount, onMounted, ref, watch } from "vue";
import { useStore } from "vuex";
import L from "leaflet";
import "leaflet/dist/leaflet.css";

const store = useStore();
const el = ref(null);

const restaurants = computed(() => store.getters["restaurants/filteredRestaurants"]);
const featured = computed(() => store.getters["restaurants/featured"]);
const center = computed(() => store.getters["location/coordinates"]);

// Theme colors ($salmon-lt, $salmon, $dark); circle markers need no image assets
const PIN = { fill: "#ff9c93", featured: "#ff6f61", stroke: "#22262D" };

// Plain variables, not refs: Leaflet objects must stay out of Vue's reactive proxies
let map = null;
let pins = null;

function drawPins() {
  pins.clearLayers();
  let featuredPin = null;
  restaurants.value.forEach((r) => {
    const isFeatured = r.id === featured.value?.id;
    const pin = L.circleMarker([r.lat, r.lon], {
      radius: isFeatured ? 11 : 7,
      color: PIN.stroke,
      weight: 2,
      fillColor: isFeatured ? PIN.featured : PIN.fill,
      fillOpacity: isFeatured ? 1 : 0.8
    })
      .bindTooltip(r.name)
      .on("click", () => store.dispatch("restaurants/selectRestaurant", r.id))
      .addTo(pins);
    if (isFeatured) featuredPin = pin;
  });
  featuredPin?.bringToFront();
}

function fitToPins() {
  const points = restaurants.value.map((r) => [r.lat, r.lon]);
  if (points.length) map.fitBounds(points, { padding: [32, 32], maxZoom: 16 });
  else map.setView([center.value.lat, center.value.lon], 14);
}

onMounted(() => {
  map = L.map(el.value, { scrollWheelZoom: false });
  L.tileLayer("https://tile.openstreetmap.org/{z}/{x}/{y}.png", {
    maxZoom: 19,
    attribution:
      '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
  }).addTo(map);
  pins = L.layerGroup().addTo(map);
  drawPins();
  fitToPins();
});

watch(restaurants, () => {
  drawPins();
  fitToPins();
});

watch(featured, (r) => {
  drawPins();
  if (r) map.panTo([r.lat, r.lon]);
});

onBeforeUnmount(() => map?.remove());
</script>

  </div>

- The illustration index moved from list position to a hash of the id (`artFor`). That way a place keeps the same art in the Featured card and the list whatever the sort or filter, which the spec's "chosen by position" wording would have broken.
